        when(metadata.get(TikaCoreProperties.INTERNAL_PATH)).thenReturn("");
        when(metadata.get(TikaCoreProperties.EMBEDDED_RELATIONSHIP_ID)).thenReturn("");
        when(metadata.get(TikaCoreProperties.EMBEDDED_RESOURCE_PATH)).thenReturn("");
        when(metadata.get(TikaCoreProperties.ORIGINAL_RESOURCE_NAME)).thenReturn("example.txt");
    }

    @Test
    public void testGetSanitizedEmbeddedFileName() throws IOException {
        FilenameUtils filenameUtils = new FilenameUtils();
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileNameWithNullPath() throws IOException {
        when(metadata.get(TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn(null);
        FilenameUtils filenameUtils = new FilenameUtils();
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals(null, result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileNameWithWhitespacePath() throws IOException {
        when(metadata.get(TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("  example.txt  ");
        FilenameUtils filenameUtils = new FilenameUtils();
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileNameWithProtocolPath() throws IOException {
        when(metadata.get(TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("http://example.com/example.txt");
        FilenameUtils filenameUtils = new FilenameUtils();
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileNameWithPrefixPath() throws IOException {
        when(metadata.get(TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("C:\\example.txt");
        FilenameUtils filenameUtils = new FilenameUtils();
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileNameWithColonPath() throws IOException {
        when(metadata.get(TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("example:txt");
        FilenameUtils filenameUtils = new FilenameUtils();
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileNameWithDotPath() throws IOException {
        when(metadata.get(TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn(".example.txt");
        FilenameUtils filenameUtils = new FilenameUtils();
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileNameWithLongPath() throws IOException {
        when(metadata.get(TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("example.txt".repeat(100));
        FilenameUtils filenameUtils = new FilenameUtils();
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt" + "example.txt".substring(0, 7) + "...txt", result);
    }
}
